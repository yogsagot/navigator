# Navigator's run-time dependencies.  pyte is the VT emulator behind
# `navkit.console`: it turns a child program's output into a grid of cells
# Navigator owns, which is what lets Ctrl+O paint that output as the desktop
# background instead of merely handing the screen back to the terminal.  It is
# pure Python and brings only wcwidth with it.
pyte>=0.8.2
# Pygments lexes for syntax highlighting (`navigator.lib.highlight`): the editor
# and viewer colour its tokens through the stylesheet's ::token rules, where
# DOS Navigator read its own DN.HGL.  Pure Python, no dependencies of its own.
# At least the release navigator/assets/highlight.ini was generated from
# (tools/highlight_ini.py), so every lexer it names exists.
pygments>=2.21
# watchdog tells a panel that the directory it shows changed on disk
# (`navigator.lib.system.dirwatch`, Options > File Manager > Setup's *Auto
# refresh panels*): inotify on Linux through ctypes, FSEvents on macOS, kqueue
# on the BSDs.  No dependencies of its own, and nothing compiled on Linux.
watchdog>=6
# s3 drive (`navigator.lib.drive.s3`):
boto3>=1.34
# F3's Markdown rendering (tests/test_viewer_markdown.py).
markdown-it-py>=3.0
# ... and reStructuredText (tests/test_viewer_rst.py).
docutils>=0.18
# ... and a PDF's text (tests/test_viewer_pdf.py).  Pure Python; only an
# AES-encrypted PDF would want `cryptography', and without it shows its source.
pypdf>=4
